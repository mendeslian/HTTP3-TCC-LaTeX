% Diagrama de sequencia da requisicao em HTTP/3 (Figura do capitulo 4).

% Listagem 4.1, listen 8443 quic reuseport), docker-compose.bench.yml,
% benchmarks/h2load (flag --h3).
\begin{tikzpicture}[
  part/.style={draw, rectangle, align=center, inner sep=2pt, font=\scriptsize, minimum width=3.2cm, minimum height=0.8cm},
  life/.style={draw=black!60, dashed},
  msg/.style={-{Stealth[length=2mm]}, thick},
  rmsg/.style={-{Stealth[length=2mm]}, thick, dashed},
  lbl/.style={font=\tiny, inner sep=1.5pt, fill=white, align=center},
  wrap/.style={font=\tiny, inner sep=1.5pt, fill=white, align=center, text width=4.4cm},
  phase/.style={font=\scriptsize\bfseries, align=center, text width=12.4cm, anchor=north, fill=white, inner sep=2pt},
  note/.style={font=\scriptsize\itshape, align=center, text width=12.4cm, anchor=north, fill=white, inner sep=2pt},
]

\node[part] (cli) at (0,0) {Cliente\\(h2load, \texttt{-{}-h3})};
\node[part] (ngx) at (5.6,0) {NGINX\\(\texttt{:8443} QUIC/UDP)};
\node[part] (api) at (11.2,0) {API\\(\texttt{node:http} \texttt{:3000})};

\draw[life] (cli.south) -- ([yshift=-12.6cm]cli.south);
\draw[life] (ngx.south) -- ([yshift=-12.6cm]ngx.south);
\draw[life] (api.south) -- ([yshift=-12.6cm]api.south);

\node[phase] at ([yshift=-0.35cm]ngx.south) {Handshake QUIC sobre UDP com TLS 1.3 integrado (ALPN \texttt{h3})};
\draw[msg] ([yshift=-1.45cm]cli.south) -- node[wrap, above] {Initial: ClientHello (ALPN h3)} ([yshift=-1.45cm]ngx.south);
\draw[rmsg] ([yshift=-2.45cm]ngx.south) -- node[wrap, above] {Initial/Handshake: ServerHello (h3)} ([yshift=-2.45cm]cli.south);
\draw[msg] ([yshift=-3.35cm]cli.south) -- node[lbl, above] {Finished} ([yshift=-3.35cm]ngx.south);

\node[phase] at ([yshift=-4.05cm]ngx.south) {Requisição HTTP/3 em um stream QUIC (\texttt{-m 1})};
\draw[msg] ([yshift=-5.05cm]cli.south) -- node[wrap, above] {HEADERS (stream 0): GET /bench/ping} ([yshift=-5.05cm]ngx.south);
\draw[msg] ([yshift=-6.05cm]ngx.south) -- node[lbl, above] {\texttt{proxy\_pass} HTTP/1.1} ([yshift=-6.05cm]api.south);
\draw[rmsg] ([yshift=-6.85cm]api.south) -- node[lbl, above] {200 \texttt{\{"ok":true\}}} ([yshift=-6.85cm]ngx.south);
\draw[rmsg] ([yshift=-7.65cm]ngx.south) -- node[lbl, above] {HEADERS + DATA (stream 0)} ([yshift=-7.65cm]cli.south);

\node[note] at ([yshift=-8.35cm]ngx.south) {Ciclo repetido enquanto dura o patamar de 60 s (aquecimento de 10 s descartado)};
\draw[msg] ([yshift=-9.45cm]cli.south) -- node[wrap, above] {HEADERS (stream 4): GET /bench/ping} ([yshift=-9.45cm]ngx.south);
\draw[msg] ([yshift=-10.45cm]ngx.south) -- node[lbl, above] {\texttt{proxy\_pass} HTTP/1.1} ([yshift=-10.45cm]api.south);
\draw[rmsg] ([yshift=-11.25cm]api.south) -- node[lbl, above] {200} ([yshift=-11.25cm]ngx.south);
\draw[rmsg] ([yshift=-12.05cm]ngx.south) -- node[lbl, above] {HEADERS + DATA (stream 4)} ([yshift=-12.05cm]cli.south);

\end{tikzpicture}
